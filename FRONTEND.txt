  <script>
    let currentReplies = {};

    const sampleEmail = {
      sender: "sarah.jenkins@acmecorp.com",
      subject: "URGENT: Final Approval for Q3 Marketing Strategy & Budget",
      body: "Hi Team,\n\nI hope you're having a good week.\n\nWe need to finalize the Q3 Marketing Strategy before our executive review tomorrow. Could you please review the attached slide deck and send over your final sign-off by 4:00 PM today?\n\nAlso, let's schedule a brief 15-minute sync at 10:00 AM tomorrow to prepare for the board presentation.\n\nThanks,\nSarah"
    };

    document.getElementById('sampleBtn').addEventListener('click', () => {
      document.getElementById('sender').value = sampleEmail.sender;
      document.getElementById('subject').value = sampleEmail.subject;
      document.getElementById('body').value = sampleEmail.body;
    });

    document.getElementById('emailForm').addEventListener('submit', async (e) => {
      e.preventDefault();

      const payload = {
        sender: document.getElementById('sender').value,
        subject: document.getElementById('subject').value,
        body: document.getElementById('body').value
      };

      // UI States
      document.getElementById('emptyState').classList.add('hidden');
      document.getElementById('resultsContent').classList.add('hidden');
      document.getElementById('loadingState').classList.remove('hidden');

      try {
        const response = await fetch('/api/analyze-email', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        const resData = await response.json();

        if (resData.success) {
          renderResults(resData.data);
        } else {
          alert('Analysis failed: ' + resData.error);
        }
      } catch (err) {
        alert('Failed to connect to backend server.');
      } finally {
        document.getElementById('loadingState').classList.add('hidden');
      }
    });

    function renderResults(data) {
      document.getElementById('resultsContent').classList.remove('hidden');

      // 1. Spam Banner
      const spamBanner = document.getElementById('spamBanner');
      if (data.isSpamOrPhishing) {
        spamBanner.classList.remove('hidden');
        document.getElementById('spamReason').innerText = data.spamReasoning;
      } else {
        spamBanner.classList.add('hidden');
      }

      // 2. Badges & Priority
      const priorityBadge = document.getElementById('priorityBadge');
      priorityBadge.innerText = `${data.priority} Priority`;
      
      if (data.priority === 'High') {
        priorityBadge.className = 'px-2.5 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-rose-500/20 text-rose-300 border border-rose-500/40';
      } else if (data.priority === 'Medium') {
        priorityBadge.className = 'px-2.5 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-amber-500/20 text-amber-300 border border-amber-500/40';
      } else {
        priorityBadge.className = 'px-2.5 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-emerald-500/20 text-emerald-300 border border-emerald-500/40';
      }

      document.getElementById('categoryBadge').innerText = data.category;
      document.getElementById('sentimentBadge').innerText = `Sentiment: ${data.sentiment}`;
      document.getElementById('langBadge').innerText = `Lang: ${data.detectedLanguage}`;

      // 3. Summary
      document.getElementById('summaryText').innerText = data.summary;

      // 4. Tasks
      const taskList = document.getElementById('taskList');
      taskList.innerHTML = '';
      if (data.tasks.length === 0) {
        taskList.innerHTML = '<li class="text-slate-500 italic">No explicit tasks extracted.</li>';
      } else {
        data.tasks.forEach(t => {
          taskList.innerHTML += `
            <li class="bg-slate-900/60 p-2.5 rounded border border-slate-700/50 flex flex-col gap-1">
              <span class="text-slate-200 font-medium">✓ ${t.task}</span>
              <span class="text-[10px] text-indigo-400 font-mono">Deadline: ${t.deadline}</span>
            </li>`;
        });
      }

      // 5. Events
      const eventList = document.getElementById('eventList');
      eventList.innerHTML = '';
      if (data.calendarEvents.length === 0) {
        eventList.innerHTML = '<li class="text-slate-500 italic">No events detected.</li>';
      } else {
        data.calendarEvents.forEach(e => {
          eventList.innerHTML += `
            <li class="bg-slate-900/60 p-2.5 rounded border border-slate-700/50">
              <div class="text-slate-200 font-medium">${e.title}</div>
              <div class="text-[10px] text-amber-400 font-mono mt-0.5">${e.dateTime}</div>
            </li>`;
        });
      }

      // 6. Follow-Up
      document.getElementById('followUpText').innerText = data.followUpReminder || "No specific follow-up required.";

      // 7. Replies
      currentReplies = data.smartReplies;
      setReplyTone('formal');
    }

    function setReplyTone(tone) {
      document.querySelectorAll('.tone-btn').forEach(btn => {
        btn.className = 'tone-btn text-xs px-2.5 py-1 rounded font-medium text-slate-400 hover:text-white';
      });
      document.getElementById(`btn-${tone}`).className = 'tone-btn text-xs px-2.5 py-1 rounded font-medium bg-indigo-600 text-white';

      document.getElementById('replyBox').value = currentReplies[tone] || '';
    }

    function copyReply() {
      const replyText = document.getElementById('replyBox');
      replyText.select();
      navigator.clipboard.writeText(replyText.value);
      alert('Reply copied to clipboard!');
    }
  </script>
