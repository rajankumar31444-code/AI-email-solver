        function loadPresetEmail(key) {
            if (!key || !sampleEmails[key]) return;

        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-white', 'text-brand-600', 'shadow-xs', 'border', 'border-slate-200/80');
                btn.classList.add('text-slate-600');
            });
            
            document.getElementById('content-' + tabId).classList.remove('hidden');
            const activeBtn = document.getElementById('tab-' + tabId);
            activeBtn.classList.remove('text-slate-600');
            activeBtn.classList.add('bg-white', 'text-brand-600', 'shadow-xs', 'border', 'border-slate-200/80');
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastMessage').innerText = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }

  "translatedSummary": "Summary translated into the requested target language (${targetLang}). If English, make it identical to summary."
}`;

            const userQuery = `Email Sender: ${sender}\nEmail Subject: ${subject}\nEmail Body:\n${body}\nTarget Language: ${targetLang}`;

            const apiKeyToUse = customApiKey || "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKeyToUse}`;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] },
                generationConfig: {
                    responseMimeType: "application/json"
                }
            };

            let data = null;
            let retries = 3;
            let delay = 1000;

            for (let attempt = 0; attempt < retries; attempt++) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
                    data = await response.json();
                    break;
                } catch (err) {
                    if (attempt === retries - 1) {
                        console.warn('API call failed, falling back to intelligent offline heuristic analysis.', err);
                    } else {
                        await new Promise(res => setTimeout(res, delay));
                        delay *= 2;
                    }
                }
            }

            document.getElementById('loadingState').classList.add('hidden');

            let parsedResult = null;
            if (data && data.candidates && data.candidates[0]?.content?.parts?.[0]?.text) {
                try {
                    let textResp = data.candidates[0].content.parts[0].text;
                    // Clean markdown code blocks if present
                    textResp = textResp.replace(/```json/g, '').replace(/```/g, '').trim();
                    parsedResult = JSON.parse(textResp);
                } catch (e) {
                    console.error('Failed to parse JSON from AI response:', e);
                }
            }

            // Fallback intelligent heuristic analysis if API fails or parse fails
            if (!parsedResult) {
                parsedResult = generateFallbackAnalysis(subject, body, targetLang);
            }

            currentAnalysisResult = parsedResult;
            renderAnalysis(parsedResult, targetLang);
            showToast('Email analyzed successfully!');
        }

        function generateFallbackAnalysis(subject, body, targetLang) {
            const lowerBody = body.toLowerCase();
            const isUrgent = lowerBody.includes('urgent') || lowerBody.includes('asap') || lowerBody.includes('immediately');
            const isSpam = lowerBody.includes('verify your') || lowerBody.includes('suspicious login') || lowerBody.includes('account suspended');
            const hasMeeting = lowerBody.includes('meeting') || lowerBody.includes('zoom') || lowerBody.includes('when:');

            return {
                summary: `This email regarding "${subject || 'General Inquiry'}" discusses key action items and operational next steps.`,
                priority: isUrgent ? "High" : (isSpam ? "High" : "Medium"),
                priorityReason: isUrgent ? "Contains urgent keywords requiring immediate response." : "Standard priority operational communication.",
                sentiment: isUrgent ? "Urgent" : (isSpam ? "Negative" : "Neutral"),
                spamRisk: {
                    isSpamOrPhishing: isSpam,
                    confidence: isSpam ? "High" : "Low",
                    reason: isSpam ? "Warning: Contains suspicious verification links and urgent account warnings." : "Clean inbox message."
                },
                tasks: [
                    { task: "Review email content and assess required response", deadline: isUrgent ? "Within 24 hours" : "By end of week" },
                    { task: "Follow up with sender if necessary", deadline: "As scheduled" }
                ],
                meeting: {
                    hasMeeting: hasMeeting,
                    title: hasMeeting ? (subject || "Scheduled Discussion") : null,
                    datetime: hasMeeting ? "Upcoming as specified in body" : null,
                    location: hasMeeting ? "Virtual / Meeting Room" : null
                },
                smartReplies: [
                    { tone: "Professional", reply: "Thank you for reaching out. I have received your message and am reviewing the details. I will follow up shortly." },
                    { tone: "Concise / Direct", reply: "Received, thanks. Will review and update you by tomorrow." },
                    { tone: "Friendly", reply: "Hi there! Thanks for the update. Let's touch base on this soon!" }
                ],
                translatedSummary: `[${targetLang}] Summary: This email discusses key action items regarding "${subject}".`
            };
        }

        function renderAnalysis(result, targetLang) {
            // Priority
            const priorityEl = document.getElementById('textPriority');
            const badgePriority = document.getElementById('badgePriority');
            priorityEl.innerText = result.priority;
            badgePriority.className = "px-2.5 py-1 rounded-full text-xs font-bold flex items-center space-x-1 " + 
                (result.priority === 'High' ? 'bg-rose-100 text-rose-700' : (result.priority === 'Medium' ? 'bg-amber-100 text-amber-700' : 'bg-emerald-100 text-emerald-700'));

            // Sentiment
            const badgeSentiment = document.getElementById('badgeSentiment');
            badgeSentiment.innerText = result.sentiment;
            badgeSentiment.className = "px-2.5 py-1 rounded-full text-xs font-bold " + 
                (result.sentiment === 'Urgent' ? 'bg-rose-100 text-rose-700' : (result.sentiment === 'Positive' ? 'bg-emerald-100 text-emerald-700' : 'bg-slate-100 text-slate-700'));

            // Security
            const textSecurity = document.getElementById('textSecurity');
            const badgeSecurity = document.getElementById('badgeSecurity');
            if (result.spamRisk.isSpamOrPhishing) {
                textSecurity.innerText = "Phishing / Spam Risk!";
                badgeSecurity.className = "px-2.5 py-1 rounded-full text-xs font-bold bg-rose-100 text-rose-700 flex items-center space-x-1.5";
                badgeSecurity.querySelector('i').className = "fa-solid fa-triangle-exclamation";
            } else {
                textSecurity.innerText = "Clean / Safe";
                badgeSecurity.className = "px-2.5 py-1 rounded-full text-xs font-bold bg-emerald-50 text-emerald-700 flex items-center space-x-1.5";
                badgeSecurity.querySelector('i').className = "fa-solid fa-shield-check";
            }

            // Summary
            document.getElementById('summaryText').innerText = result.summary;
            document.getElementById('priorityReasonText').innerText = result.priorityReason;

            // Translated Summary
            const transContainer = document.getElementById('translatedSummaryContainer');
            if (targetLang && targetLang !== 'English' && result.translatedSummary) {
                document.getElementById('targetLangLabel').innerText = targetLang;
                document.getElementById('translatedSummaryText').innerText = result.translatedSummary;
                transContainer.classList.remove('hidden');
            } else {
                transContainer.classList.add('hidden');
            }

            // Tasks
            const tasksContainer = document.getElementById('tasksListContainer');
            tasksContainer.innerHTML = '';
            const taskBadge = document.getElementById('taskBadge');
            if (result.tasks && result.tasks.length > 0) {
                taskBadge.innerText = result.tasks.length;
                taskBadge.classList.remove('hidden');
                result.tasks.forEach((t, idx) => {
                    const div = document.createElement('div');
                    div.className = "bg-white p-3.5 rounded-xl border border-slate-200 shadow-xs flex items-start space-x-3";
                    div.innerHTML = `
                        <input type="checkbox" id="task-${idx}" class="mt-1 h-4 w-4 text-brand-600 rounded border-slate-300 focus:ring-brand-500">
                        <div class="flex-1">
                            <label for="task-${idx}" class="text-xs font-medium text-slate-800 block">${t.task}</label>
                            <span class="text-[11px] text-slate-400 mt-0.5 inline-flex items-center space-x-1">
                                <i class="fa-regular fa-clock"></i>
                                <span>Deadline: ${t.deadline}</span>
                            </span>
                        </div>
                    `;
                    tasksContainer.appendChild(div);
                });
            } else {
                taskBadge.classList.add('hidden');
                tasksContainer.innerHTML = '<p class="text-xs text-slate-400 text-center py-6">No explicit tasks extracted.</p>';
            }

            // Calendar / Meeting
            const meetingContainer = document.getElementById('meetingDetailsContainer');
            if (result.meeting && result.meeting.hasMeeting) {
                meetingContainer.innerHTML = `
                    <div class="flex items-start space-x-3 bg-brand-50/50 p-4 rounded-xl border border-brand-100">
                        <div class="w-10 h-10 rounded-lg bg-brand-100 text-brand-600 flex items-center justify-center shrink-0">
                            <i class="fa-solid fa-calendar-check text-base"></i>
                        </div>
                        <div class="space-y-1 flex-1">
                            <h5 class="text-xs font-bold text-slate-800">${result.meeting.title}</h5>
                            <p class="text-xs text-slate-600"><i class="fa-regular fa-clock mr-1 text-slate-400"></i> ${result.meeting.datetime}</p>
                            <p class="text-xs text-slate-600"><i class="fa-solid fa-location-dot mr-1 text-slate-400"></i> ${result.meeting.location}</p>
                        </div>
                    </div>
                    <button onclick="showToast('Meeting event added to calendar simulation!')" class="w-full bg-slate-900 hover:bg-slate-800 text-white text-xs font-medium py-2.5 rounded-xl transition-colors flex items-center justify-center space-x-2">
                        <i class="fa-solid fa-calendar-plus"></i>
                        <span>Add to Calendar</span>
                    </button>
                `;
            } else {
                meetingContainer.innerHTML = `
                    <div class="text-center py-8">
                        <div class="w-12 h-12 rounded-full bg-slate-100 text-slate-400 flex items-center justify-center mx-auto mb-2">
                            <i class="fa-regular fa-calendar-xmark text-lg"></i>
                        </div>
                        <p class="text-xs font-medium text-slate-600">No meetings or calendar events detected in this email.</p>
                    </div>
                `;
            }

            // Smart Replies
            const repliesContainer = document.getElementById('smartRepliesContainer');
            repliesContainer.innerHTML = '';
            if (result.smartReplies && result.smartReplies.length > 0) {
                result.smartReplies.forEach((r, idx) => {
                    const div = document.createElement('div');
                    div.className = "bg-white p-4 rounded-xl border border-slate-200 shadow-xs space-y-2.5";
                    div.innerHTML = `
                        <div class="flex items-center justify-between">
                            <span class="text-[11px] font-bold text-brand-600 uppercase tracking-wider bg-brand-50 px-2 py-0.5 rounded">${r.tone}</span>
                            <button onclick="copyReplyText(${idx})" class="text-xs text-slate-500 hover:text-brand-600 font-medium flex items-center space-x-1">
                                <i class="fa-regular fa-copy"></i>
                                <span>Copy Reply</span>
                            </button>
                        </div>
                        <p id="reply-text-${idx}" class="text-xs text-slate-700 bg-slate-50 p-3 rounded-lg border border-slate-100 leading-relaxed">${r.reply}</p>
                    `;
                    repliesContainer.appendChild(div);
                });
            }

            // Enable Voice button
            document.getElementById('voiceBtn').classList.remove('hidden');
            switchTab('overview');
        }

        function copyReplyText(idx) {
            const el = document.getElementById(`reply-text-${idx}`);
            if (!el) return;
            const text = el.innerText;
            const textarea = document.createElement('textarea');
            textarea.value = text;
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                showToast('Reply copied to clipboard!');
            } catch (err) {
                showToast('Failed to copy text.');
            }
            document.body.removeChild(textarea);
        }

        function playVoiceSummary() {
            if (!currentAnalysisResult || !currentAnalysisResult.summary) {
                showToast('No summary available to read.');
                return;
            }
            if (!('speechSynthesis' in window)) {
                showToast('Speech synthesis not supported in this browser.');
                return;
            }
            window.speechSynthesis.cancel();
            const utterance = new SpeechSynthesisUtterance(currentAnalysisResult.summary);
            utterance.rate = 1.0;
            utterance.pitch = 1.0;
            window.speechSynthesis.speak(utterance);
            showToast('Playing voice summary...');
        }

        function exportTasks() {
            if (!currentAnalysisResult || !currentAnalysisResult.tasks || currentAnalysisResult.tasks.length === 0) {
                showToast('No tasks to export.');
                return;
            }
            let content = "AI EMAIL ASSISTANT - EXPORTED TASKS\n=====================================\n\n";
            currentAnalysisResult.tasks.forEach((t, i) => {
                content += `${i + 1}. [ ] ${t.task}\n   Deadline: ${t.deadline}\n\n`;
            });
            const blob = new Blob([content], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'email-tasks.txt';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            showToast('Tasks exported successfully as text file!');
        }
